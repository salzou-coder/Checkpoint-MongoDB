MongoDB Checkpoint – CRUD Operations

1. Display all contacts
db.contactlist.find()

2. Display one contact by ID
db.contactlist.find({ _id: ObjectId("XXXXXXXXXXXX") })

3. Contacts with age > 18
db.contactlist.find({ age: { $gt: 18 } })

4. Contacts with age > 18 and name containing "ah"
db.contactlist.find({
  age: { $gt: 18 },
  $or: [
    { firstName: { $regex: "ah", $options: "i" } },
    { lastName: { $regex: "ah", $options: "i" } }
  ]
})

5. Update first name
db.contactlist.updateOne(
  { firstName: "Seif", lastName: "Kefi" },
  { $set: { firstName: "Anis" } }
)

6. Delete contacts under age 5
db.contactlist.deleteMany({ age: { $lt: 5 } })

7. Display all contacts again
db.contactlist.find()
